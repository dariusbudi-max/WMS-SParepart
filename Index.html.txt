      }

    computed: {
      // 🔹 Filter master items by search query
      filteredMasterItems() {
        if (!this.masterItems || !Array.isArray(this.masterItems)) return [];
        if (!this.masterSearch) return this.masterItems;

        const q = this.masterSearch.toLowerCase();
        return this.masterItems.filter(i =>
          String(i.nama || '').toLowerCase().includes(q) ||
          String(i.kode || '').toLowerCase().includes(q) ||
          String(i.lokasi || '').toLowerCase().includes(q)
        );
      },

      // 🔹 Sorted filtered master items
      sortedFilteredItems() {
        if (!this.filteredMasterItems || !Array.isArray(this.filteredMasterItems)) return [];

        const items = [...this.filteredMasterItems];
        const key = this.sortConfig?.key || '';
        const dir = this.sortConfig?.direction === 'asc' ? 1 : -1;

        if (!key) return items;

        return items.sort((a, b) => {
          const valA = a[key];
          const valB = b[key];

          if (['stok', 'stokMin'].includes(key)) {
            return (Number(valA || 0) - Number(valB || 0)) * dir;
          }

          return String(valA || '').localeCompare(String(valB || '')) * dir;
        });
      },

      categoryList() {
        return Array.isArray(this.allCategories) ? this.allCategories : [];
      },

      // 🔹 Filter by category & search
      filteredSearchResults() {
        // renamed from searchResults
        return this.masterItems.filter(item => {
          const matchCategory =
            this.filterCategory === 'ALL' ||
            item.kategori === this.filterCategory;

          const matchSearch =
            item.nama.toLowerCase().includes(this.searchQuery.toLowerCase()) ||
            item.kode.toLowerCase().includes(this.searchQuery.toLowerCase());

          return matchCategory && matchSearch;
        });
      },

      // 🔹 Filter recent transactions for dashboard
      filteredHistory() {
        if (!this.dashData?.recentTx || !Array.isArray(this.dashData.recentTx)) return [];

        const q = (this.searchQuery || '').toLowerCase();
        return this.dashData.recentTx.filter(tx => {
          const user = (tx.user || '').toLowerCase();
          const kode = (tx.kode || '').toLowerCase();
          const nama = (tx.nama || '').toLowerCase();
          const dept = (tx.dept || '').toLowerCase();
          return user.includes(q) || kode.includes(q) || nama.includes(q) || dept.includes(q);
        });
      },

      // 🔹 Filter recent transactions for cancellation page
      filteredCancelTx() {
        if (!this.recentTx || !Array.isArray(this.recentTx) || this.recentTx.length === 0) return [];

        const q = (this.searchCancelTx || '').toLowerCase().trim();
        if (!q) return this.recentTx;

        return this.recentTx.filter(tx => {
          const kode = (tx.kode || '').toLowerCase();
          const nama = (tx.nama || '').toLowerCase();
          return kode.includes(q) || nama.includes(q);
        });
      },

      // 🔹 Paginate history (limit 300)
      paginatedHistory() {
        return (this.filteredHistory || []).slice(0, 300);
      },

      // 🔹 Safe clone of recent transactions
      recentTxSafe() {
        if (!this.recentTx || !Array.isArray(this.recentTx)) return [];
        return this.recentTx.map(tx => Object.freeze({ ...tx }));
      },

      // 🔹 Filter departments by search query
      filteredDepts() {
        if (!this.departments || !Array.isArray(this.departments)) return [];
        const q = (this.deptSearch || '').toLowerCase();
        return this.departments.filter(d => String(d).toLowerCase().includes(q));
      },

      // 🔹 Filter departments for reports
      filteredDeptsReport() {
        if (!this.departments || !Array.isArray(this.departments)) return [];

        const search = (this.dashFilter?.dept || '').toLowerCase();
        if (!search) return this.departments.slice(0, 15);
        return this.departments.filter(d => String(d).toLowerCase().includes(search));
      },
    },

    methods: {
      async restoreSession() {
        const token = localStorage.getItem("wms_v5_final");
        const savedUser = localStorage.getItem("wms_v5_user");

        if (!token || !savedUser) {
          this.forceLogout(false);
          return;
        }

        try {
          const parsedUser = JSON.parse(savedUser);

          // 🔐 VALIDASI TOKEN KE BACKEND
          const session = await this.runGS("validateSession", token);

          if (!session || !session.valid) {
            this.forceLogout(false);
            return;
          }

          // ✅ Gunakan data dari backend sebagai source of truth
          this.user = {
            token: token,
            nama: session.nama,
            role: session.role
          };

          this.isLoggedIn = true;

          this.initAfterLogin();
          this.loadInitialDataByRole();
          this.startSessionWatcher();
          this.setupClickOutside();

          this.$nextTick(() => {
            this.$refs.myInput?.focus();
          });

        } catch (err) {
          console.warn("Restore session gagal:", err);
          this.forceLogout(false);
        }
      },

      startSessionWatcher() {
        this.sessionInterval = setInterval(async () => {
          const token = localStorage.getItem("wms_v5_final");
          if (!token) {
            this.forceLogout(true);
            return;
          }

          const session = await this.runGS("validateSession", token);

          if (!session || !session.valid) {
            this.forceLogout(true);
          }
        }, 300000); // 5 menit
      },

      // 🔹 Wrapper aman untuk backend
      async safeRunGS(method, payload = {}) {
        if (!this.user?.token) {
          Swal.fire('Error', 'User belum login atau token kosong', 'error');
          return { status: 'error', message: 'Token tidak tersedia', data: [] };
        }

        // Pastikan payload selalu objek, tambahkan token
        const sendPayload = { ...payload, token: this.user.token };

        try {
          const res = await this.runGS(method, sendPayload);

          console.log(`Response from ${method}:`, res);

          if (!res) return { status: 'error', message: 'Tidak ada response dari server', data: [] };
          if (Array.isArray(res)) return { status: 'success', data: res };
          if (res.status === 'success') {
            if (!res.data) res.data = [];
            return res;
          }
          return res;
        } catch (err) {
          console.error(`Error in safeRunGS (${method}):`, err);
          return { status: 'error', message: 'Terjadi kesalahan server', data: [] };
        }
      },

      // 🔹 Ambil recent transactions + update dashboard otomatis
      async fetchRecentTx() {
        this.isLoadingTx = true;
        try {
          // 🔐 Pastikan token string tersedia
          const token = this.user?.token;
          if (!token) {
            Swal.fire('Error', 'User belum login', 'error');
            this.recentTx = [];
            return;
          }

          // Panggil backend dengan token
          const res = await this.runGS('getRecentTransactions', token);

          if (res.status === 'success') {
            // Ambil data transaksi atau array kosong
            this.recentTx = Array.isArray(res.data) ? res.data : [];
          } else {
            console.warn('fetchRecentTx backend error:', res.message);
            Swal.fire('Gagal', res.message || 'Terjadi kesalahan', 'error');
            this.recentTx = [];
          }

          // Refresh dashboard jika fungsi tersedia
          if (typeof this.fetchDashboard === 'function') this.fetchDashboard();

        } catch (err) {
          console.error('fetchRecentTx error:', err);
          this.recentTx = [];
          Swal.fire('Error', 'Gagal memuat transaksi', 'error');
        } finally {
          this.isLoadingTx = false;
        }
      },

      // 🔹 Batalkan transaksi & live update recentTx + dashboard
      async doCancelTx(tx) {
        if (!this.user?.token) {
          Swal.fire('Error', 'User belum login', 'error');
          return;
        }

        try {
          const confirm = await Swal.fire({
            title: 'Batalkan Transaksi?',
            html: `Anda yakin ingin membatalkan transaksi <b>${tx.jenis}</b> untuk <b>${tx.nama}</b> (Qty: ${tx.qty})?<br><br><span class="text-sm text-rose-500">Stok akan otomatis dikembalikan.</span>`,
            icon: 'warning',
            showCancelButton: true,
            confirmButtonColor: '#e11d48',
            confirmButtonText: 'Ya, Batalkan!'
          });

          if (!confirm.isConfirmed) return;

          Swal.fire({ title: 'Memproses...', allowOutsideClick: false, didOpen: () => Swal.showLoading() });

          // Panggil backend dengan token
          const res = await this.runGS('cancelTransaction', this.user.token, tx.rowId);

          Swal.close();

          if (res.status === 'success') {
            Swal.fire('Berhasil', res.message, 'success');

            // Update live recentTx
            const index = this.recentTx.findIndex(t => t.rowId === tx.rowId);
            if (index !== -1) this.recentTx[index].ket = `[DIBATALKAN] ${this.recentTx[index].ket}`;

            if (typeof this.fetchDashboard === 'function') this.fetchDashboard();
          } else {
            Swal.fire('Gagal', res.message || 'Terjadi kesalahan', 'error');
          }
        } catch (err) {
          console.error('doCancelTx error:', err);
          Swal.fire('Error', err.message || 'Terjadi kesalahan', 'error');
        }
      },

             Swal.fire('Opps', 'Department wajib diisi!', 'warning');
             return;
         }
         
         const payload = { items: this.cart, jenis: this.txType, dept: this.txDept, user: this.user.nama, ket: this.txNote };
         
         let confirmText = `Proses ${this.txType}?`;
         if(this.txType === 'RETURN') confirmText = 'Barang akan dikembalikan ke Stok?';
         
         Swal.fire({
            title: 'Konfirmasi', 
            text: confirmText,
            icon: 'question',
            showCancelButton:true, 
            confirmButtonColor: '#2563eb',
            confirmButtonText:'Ya, Proses'
         }).then(r=>{
            if(r.isConfirmed) this.runGS('prosesTransaksiMulti', payload).then(res => {
               if(res.status==='success') { 
                   Swal.fire('Berhasil',res.message,'success'); 
                   this.cart=[]; this.txNote=''; this.searchQuery=''; this.searchResults=[];
                   this.fetchDashboard(); this.txDept = '';
               }
               else Swal.fire('Gagal',res.message,'error');
            });
         });
      },

      handleScan() {
        if (this.page !== 'transaksi') return;

        if (!Array.isArray(this.masterItems) || this.masterItems.length === 0) {
          Swal.fire('Tunggu', 'Data barang belum siap', 'info');
          return;
        }

        const code = (this.searchQuery || '').trim();
        if (!code) return;

        const now = Date.now();
        if (now - this.lastScanTime < 300) return;
        this.lastScanTime = now;

        const item = this.masterItems.find(i =>
          String(i.kode).toLowerCase() === code.toLowerCase()
        );

        if (!item) {
          Swal.fire('404', 'Barang tidak ditemukan', 'warning');
          this.resetScan();
          return;
        }

        this.addToCart(item, this.scanQty || 1);
        this.resetScan();
      },

        this.scanQty = 1

      async openScanner() {
        if (this.html5QrcodeScanner) return;
        this.showScanner = true;

        this.$nextTick(async () => {
          const readerEl = document.getElementById("reader");
          if (!readerEl) return console.warn("Element #reader tidak ditemukan");

          const getBoxSize = () => {
            const maxBox = 300;
            const size = Math.min(
              maxBox,
              readerEl.offsetHeight * 0.9,
              readerEl.offsetWidth * 0.9
            );
            return Math.max(size, 50);
          };

          const onScan = (decodedText) => {
            if (this.isScanLocked) return;
            this.isScanLocked = true;

            this.scanSuccessFlash = true;
            setTimeout(() => (this.scanSuccessFlash = false), 300);

            console.log("Scanned:", decodedText);

            setTimeout(() => (this.isScanLocked = false), 1000);
          };

          try {
            this.html5QrcodeScanner = new Html5Qrcode("reader");

            // Ambil semua kamera
            this.cameras = await Html5Qrcode.getCameras();
            if (!this.cameras || this.cameras.length === 0) {
              throw new Error("Camera tidak ditemukan");
            }

            // Pilih kamera belakang dulu
            const backCamera = this.cameras.find((d) =>
              d.label.toLowerCase().includes("back") ||
              d.label.toLowerCase().includes("rear") ||
              d.label.toLowerCase().includes("environment")
            );
            this.activeCameraId = backCamera ? backCamera.id : this.cameras[0].id;

            await this.html5QrcodeScanner.start(
              this.activeCameraId,
              {
                fps: 10,
                qrbox: getBoxSize(),
                aspectRatio: 1,
              },
              onScan
            );

            // Resize handler
            this._resizeHandler = async () => {
              if (!this.html5QrcodeScanner || this._isResizing) return;
              this._isResizing = true;
              try {
                await this.html5QrcodeScanner.stop();
                await this.html5QrcodeScanner.start(
                  this.activeCameraId,
                  {
                    fps: 10,
                    qrbox: getBoxSize(),
                    aspectRatio: 1,
                  },
                  onScan
                );
              } catch (err) {
                console.warn("Resize restart gagal:", err);
              }
              this._isResizing = false;
            };
            window.addEventListener("resize", this._resizeHandler);

          } catch (err) {
            console.error("Error kamera:", err);
            alert("Tidak dapat mengakses kamera. Pastikan HTTPS dan izin kamera aktif.");
            this.showScanner = false;
            this.html5QrcodeScanner = null;
          }
        });
      },

      async switchCamera() {
        if (!this.html5QrcodeScanner || this.cameras.length < 2) return;

        const currentIndex = this.cameras.findIndex(d => d.id === this.activeCameraId);
        const nextIndex = (currentIndex + 1) % this.cameras.length;
        this.activeCameraId = this.cameras[nextIndex].id;

        try {
          await this.html5QrcodeScanner.stop();
          await this.html5QrcodeScanner.start(
            this.activeCameraId,
            { fps: 10, qrbox: 250, aspectRatio: 1 },
            (decodedText) => {
              if (this.isScanLocked) return;
              this.isScanLocked = true;
              this.scanSuccessFlash = true;
              setTimeout(() => (this.scanSuccessFlash = false), 300);
              console.log("Scanned:", decodedText);
              setTimeout(() => (this.isScanLocked = false), 1000);
            }
          );
        } catch (err) {
          console.warn("Switch camera gagal:", err);
        }
      },

      async closeScanner() {
        if (this._resizeHandler) {
          window.removeEventListener("resize", this._resizeHandler);
          this._resizeHandler = null;
        }

        if (this.html5QrcodeScanner) {
          try {
            const state = this.html5QrcodeScanner.getState?.();
            if (state === 1 || state === 2) await this.html5QrcodeScanner.stop();
            await this.html5QrcodeScanner.clear();
          } catch (err) {
            console.warn("Stop normal gagal, lanjut force stop:", err);
          }

          try {
            const videoEl = document.querySelector("#reader video");
            if (videoEl && videoEl.srcObject) {
              videoEl.srcObject.getTracks().forEach(track => track.stop());
              videoEl.srcObject = null;
            }
          } catch (err) {
            console.warn("Force stop video gagal:", err);
          }

          const readerEl = document.getElementById("reader");
          if (readerEl) readerEl.innerHTML = "";

          this.html5QrcodeScanner = null;
          this.activeCameraId = null;
        }

        this.isScanLocked = false;
        this.scanSuccessFlash = false;
        this.showScanner = false;
        this._isResizing = false;
      },

      fetchUsers() {
        if (!this.user || !this.user.token) {
          console.warn('User belum login, token kosong');
          return;
        }

        this.runGS('adminGetAllUsers', this.user.token)
          .then(res => {
            if (res && !res.status) { 
              this.usersList = res; 
            } else if (res && res.status === 'error') {
              Swal.fire('Error', res.message, 'error');
            }
          })
          .catch(err => {
            console.error("Gagal mengambil user:", err);
          });
      },

      manageUser(username, action, rolePayload) {
        if (!username || !action) {
          Swal.fire('Error', 'Parameter tidak lengkap', 'error');
          return;
        }

        const token = localStorage.getItem('wms_v5_final');
        if (!token) {
          Swal.fire('Session Expired', 'Silakan login ulang', 'warning');
          return;
        }

        Swal.fire({
          title: 'Memproses...',
          allowOutsideClick: false,
          didOpen: () => Swal.showLoading()
        });

        google.script.run
          .withSuccessHandler(res => {
            Swal.close();

            if (!res || typeof res !== 'object') {
              Swal.fire('Error', 'Response tidak valid dari server', 'error');
              return;
            }

            if (res.status === 'success') {
              Swal.fire('Berhasil', res.message || 'Berhasil diproses', 'success');
              this.fetchUsers();
            } else {
              Swal.fire('Gagal', res.message || 'Terjadi kesalahan', 'error');
            }
          })
          .withFailureHandler(err => {
            Swal.close();
            Swal.fire('System Error', err?.message || err.toString(), 'error');
          })
          .adminManageUser(token, username, action, rolePayload);
      },

      },

        this.showChat = false;
        this.stopChatInterval();

        if (this.chatInterval) {
          clearInterval(this.chatInterval);
          this.chatInterval = null;

      // Ambil semua history chat
      fetchChat() {
        google.script.run.withSuccessHandler(res => {
          const newMessages = res || [];
          this.chatMessages = newMessages;
          this.lastChatCount = newMessages.length;
          this.scrollToBottom();
        }).getChatMessages();
      },

      // Cek pesan baru
      fetchChatLight() {
        google.script.run.withSuccessHandler(res => {
          const newMessages = res || [];
          if (newMessages.length > this.lastChatCount) {
            // Tandai pesan baru
            const newCount = newMessages.length - this.lastChatCount;
            this.chatMessages = newMessages.map((msg, i) => ({
              ...msg,
              isNew: i >= this.lastChatCount // flag untuk highlight
            }));
            this.lastChatCount = newMessages.length;
            this.scrollToBottom(true);
          }
        }).getChatMessages();
      },

      // Scroll otomatis, optional highlight
      scrollToBottom(highlight = false) {
        this.$nextTick(() => {
          const container = this.$refs.chatContainer;
          if (container) {
            container.scrollTop = container.scrollHeight;
            if (highlight) {
              // hilangkan flag isNew setelah 2 detik
              setTimeout(() => {
                this.chatMessages = this.chatMessages.map(msg => ({ ...msg, isNew: false }));
              }, 2000);
            }
          }
        });
      },

      // Kirim pesan baru
      sendMessage() {
          const message = (this.newMessage || '').trim();
          if (!message) return;

          const sender = this.user?.nama || "Guest"; // 🔹 perbaikan di sini
          google.script.run.withSuccessHandler(res => {
              if (res?.status === 'success') {
                  this.newMessage = '';
                  this.fetchChatLight(); // refresh chat setelah kirim
              } else {
                  Swal.fire('Error', res.message || 'Gagal mengirim pesan', 'error');
              }
          }).sendChatMessage(sender, message);
      },
